   118	    }

   121	    if (prev) await deleteObjects([previewKey(id, prev)]);

   229	export async function upsertMessage(
   230	  message: IncomingMessage,
   231	  select: Select,
   232	) {
   233	  const database = getPostgres();
   234	  let messageCountQuery = database.selectFrom('Messages').select(database.fn.countAll().as('count'));
   235	  if (isUserSelect(select)) {
   236	    messageCountQuery = messageCountQuery.where('fk_users_id', '=', select.dbUserId);
   237	  } else {
   238	    const guildId = await upsertGuild(select.guild.id);
   239	    messageCountQuery = messageCountQuery.where('fk_guilds_id', '=', guildId);
   240	  }
   241	
   242	  const { count: messageCount } = await messageCountQuery.executeTakeFirstOrThrow();
   243	  if (!message.id && Number(messageCount) >= MAX_MESSAGES) {
   244	    throw new TRPCError({
   245	      code: 'BAD_REQUEST',
   246	      message: `You have reached the limit of ${MAX_MESSAGES} messages per ${isUserSelect(select) ? 'user' : 'guild'}.`,
   247	    });
   248	  }
   249	
   250	  const values: InsertObject<DatabaseSchema, 'Messages'> = {
   251	    ...buildMessageValues(message, select),
   252	    fk_guilds_id: isUserSelect(select) ? null : await upsertGuild(select.guild.id),
   253	  };
   254	
   255	  if (message.id) values.id = message.id;
   256	
   257	  const { id, savedAtMicros } = await buildUpsertMessageQuery(database, values)
   258	    .executeTakeFirstOrThrow();
   259	
   260	  const newMessagePromise = getMessage(id);
   261	
   262	  waitUntil(runPreviewContinuation({ id, nonce: mintNonce(savedAtMicros), message: message.data }));
   263	
   264	  message && hashMessage(message);
   265	  if (message.id) {
   266	    const pusher = new Pusher({
   267	      appId: env.PUSHER_APP_ID,
   268	      key: env.NEXT_PUBLIC_PUSHER_KEY,
   269	      secret: env.PUSHER_SECRET,
   270	      cluster: env.NEXT_PUBLIC_PUSHER_CLUSTER,
   271	      useTLS: true,
   272	    });
   273	
   274	    await pusher.trigger(genMessageChannelKey(message.id), 'message-update', JSON.stringify(hashMessage(message)));
   275	  }
   276	
   277	  return newMessagePromise;
   278	}
