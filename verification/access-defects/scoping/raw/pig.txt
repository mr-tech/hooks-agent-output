    15	function processImage(req: IncomingMessage, res: ServerResponse) {
    16	  let body = '';
    17	  req.on('data', (chunk: Buffer) => {

    19	  });
    20	
    21	  req.on('end', () => {
    22	    let parsedBody: { id: string, nonce: string, message: Record<string, unknown> };
    23	    try {
    24	      parsedBody = JSON.parse(body) as typeof parsedBody;
    25	      if (typeof parsedBody !== 'object' || parsedBody === null) throw new TypeError('Request body is not an object.');
    26	    } catch (err) {
    27	      res.writeHead(500);
    28	      res.end();
    29	      console.error(err);
    30	      return;
    31	    }

    33	    const start = Date.now();

    35	      console.debug(`Generated image for message ${id} with nonce ${nonce} in ${Date.now() - start}ms`);
    36	      res.writeHead(204);
    37	      res.end();
    38	    }).catch((err) => {
    39	      if (err instanceof PreviewFailure) console.error(`preview-image-failure id=${id} class=${err.failureClass} elapsedMs=${Date.now() - start}`);

    41	      res.end();
    42	      console.error(err);
    43	    });
    44	  });
    45	}

    25	async function generateMessagePreviewImage(browser: Browser, data: Record<string, unknown>) {
    26	  const page = await browser.newPage();
    27	
    28	  try {
    29	    await page.setViewport({ width: 720, height: 840, deviceScaleFactor: 1 / 3.75 });
    30	
    31	    const url = new URL(config.PREVIEW_ENDPOINT);
    32	    url.searchParams.append('data', JSON.stringify(data));
    33	
    34	    const response = await page.goto(url.toString(), { waitUntil: 'networkidle0' })
    35	      .catch((err) => { throw new PreviewFailure('goto-rejected', err); });
    36	    const status = response?.status() ?? 0;
    37	    if (status < 200 || status > 299) throw new PreviewFailure('error-status');

    39	    return await page.screenshot({
    40	      type: 'jpeg',
    41	      quality: 90,
    42	    }).catch((err) => { throw new PreviewFailure('screenshot', err); });
    43	  } finally {
    44	    await page.close().catch((err) => { console.error(err); });
    45	  }

     7	interface Browser extends BrowserBase { _useCount: number, _id: number }

     9	interface Config extends GenericPoolOptions {
    10	  max?: number;
    11	  min?: number;
    12	  idleTimeoutMillis?: number;
    13	  maxUses?: number;
    14	  testOnBorrow?: boolean;
    15	  puppeteerLaunchOptions?: PuppeteerLaunchOptions;
    16	  validator?: (instance: Browser) => Promise<boolean>;
    17	}
    18	
    19	type WaitingQueue = { dequeue?: () => undefined | { reject: (err: unknown) => void } };
    20	const waitingQueueDequeue = (pool: Pool<Browser>) => {
    21	  const queue = (pool as unknown as { _waitingClientsQueue?: WaitingQueue })._waitingClientsQueue;
    22	  return queue?.dequeue?.bind(queue);
    23	};
    24	
    25	let currentId = 0;
    26	const createPool = ({
    27	  max = 10,
    28	  // optional. if you set this, make sure to drain() (see step 3)
    29	  min = 2,
    30	  // specifies how long a resource can stay idle in pool before being removed
    31	  idleTimeoutMillis = 30000,
    32	  // specifies the maximum number of times a resource can be reused before being destroyed
    33	  maxUses = 50,
    34	  testOnBorrow = true,
    35	  puppeteerLaunchOptions = {},
    36	  validator = () => Promise.resolve(true),
    37	  ...otherConfig
    38	}: Config = {}) => {
    39	  const poolRef = { pool: null as unknown as Pool<Browser> };
    40	  // TODO: randomly destroy old instances to avoid resource leak?
    41	  const factory = {
    42	    create: async () => {
    43	      try {
    44	        const browser = await puppeteer.launch(puppeteerLaunchOptions);
    45	        const tmp = browser as Browser;
    46	        tmp._useCount = 0;
    47	        tmp._id = ++currentId;
    48	        console.debug(`Created instance ${tmp._id}. Active instances: ${poolRef.pool.size}.`);
    49	        return tmp;
    50	      } catch (err) {
    51	        console.error(err);
    52	        throw err;
    53	      }
    54	    },
    55	
    56	    destroy: (instance: Browser) => {
    57	      console.debug(`Destroying instance ${instance._id}. Active instances: ${poolRef.pool.size}.`);
    58	      return instance.close();
    59	    },
    60	    validate: (instance: Browser) => validator(instance)
    61	      .then((valid) => Promise.resolve(valid && (maxUses <= 0 || instance._useCount < maxUses))),
    62	  };
    63	  const config: GenericPoolOptions = {
    64	    max,
    65	    min,
    66	    idleTimeoutMillis,
    67	    testOnBorrow,
    68	    ...otherConfig,
    69	  };
    70	  const pool = genericPool.createPool<Browser>(factory as Factory<Browser>, config);
    71	  poolRef.pool = pool;
    72	
    73	  const genericAcquire = pool.acquire.bind(pool);
    74	  const genericRelease = pool.release.bind(pool);
    75	  pool.acquire = () => genericAcquire().then((instance) => {
    76	    console.debug(`Acquired instance ${instance._id}.`);
    77	    instance._useCount += 1;
    78	    return instance;
    79	  });
    80	
    81	  pool.release = (instance: Browser) => {
    82	    console.debug(`Releasing instance ${instance._id}.`);
    83	    return genericRelease(instance);
    84	  };
    85	  pool.use = async (fn) => {
    86	    const resource = await pool.acquire();
    87	    try {
    88	      return await fn(resource);
    89	    } finally {
    90	      await pool.release(resource);
    91	    }
    92	  };
    93	
    94	  const dequeueWaiting = waitingQueueDequeue(pool);
    95	  if (dequeueWaiting) pool.on('factoryCreateError', (err) => { dequeueWaiting()?.reject(err); });
    96	
    97	  return pool;
    98	};
