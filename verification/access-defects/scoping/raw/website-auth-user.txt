    60	async function revokeAccessToken(accessToken: string) {
    61	  const myHeaders = new Headers();
    62	  myHeaders.append('Content-Type', 'application/x-www-form-urlencoded');
    63	
    64	  const urlencoded = new URLSearchParams();
    65	  urlencoded.append('client_id', env.NEXT_PUBLIC_DISCORD_CLIENT_ID);
    66	  urlencoded.append('client_secret', env.DISCORD_CLIENT_SECRET);
    67	  urlencoded.append('token', accessToken);
    68	
    69	  await fetch('https://discord.com/api/v10/oauth2/token/revoke', {
    70	    method: 'POST',
    71	    headers: myHeaders,
    72	    body: urlencoded,
    73	  }).catch(() => {});
    74	}

    76	async function removeUserSession(sessionToken: string, userDbId: string) {
    77	  let account;
    78	  try {
    79	    account = await getPostgres()
    80	      .selectFrom('Accounts')
    81	      .select(['access_token', 'providerAccountId'])
    82	      .where('userId', '=', userDbId)
    83	      .executeTakeFirst();
    84	  } catch (err) {
    85	    await redis.clearUser(userDbId, sessionToken).catch(() => {});
    86	    throw err;
    87	  }
    88	
    89	  if (!account) {
    90	    await redis.clearUser(userDbId, sessionToken);
    91	    return;
    92	  }
    93	
    94	  if (account.access_token) await revokeAccessToken(account.access_token);
    95	
    96	  await Promise.all([
    97	    redis.clearUser(userDbId, sessionToken, account.providerAccountId),
    98	    getPostgres()
    99	      .updateTable('Accounts')
   100	      .set({ access_token: null, refresh_token: null })
   101	      .where('provider', '=', 'discord')
   102	      .where('providerAccountId', '=', account.providerAccountId)
   103	      .execute(),
   104	  ]);
   105	}

   112	async function authCallback({ session, user }: AuthCallbackParams): Promise<NextAuthSession> {
   113	  if (session.user) {
   114	    const account = await redis.fetchUserAccount(user.id as unknown as string);
   115	
   116	    if (!account) {
   117	      // @ts-expect-error - `{}` is a valid return for unauthenticated users. client will handle de-authenticating.
   118	      return {};
   119	    }
   120	
   121	    const { access_token, providerAccountId } = account;
   122	
   123	    const guilds = access_token ? await redis.fetchUserGuilds(providerAccountId, access_token) : [];
   124	
   125	    session.user.id = providerAccountId;
   126	    session.user.guilds = guilds;
   127	  }
   128	
   129	  return session;
   130	}

   132	async function hydratedAuthCallback(params: AuthCallbackParams): Promise<NextAuthSession> {
   133	  const { session, user } = params;
   134	
   135	  if (session.user) {
   136	    const { preferences } = user;
   137	
   138	    const messagePromise = Promise.resolve(preferences?.recentMessageIds
   139	      ? getUserRecentMessages(preferences.recentMessageIds)
   140	      : []);
   141	    messagePromise.catch(() => {});
   142	
   143	    const userDbId = user.id;
   144	    const account = await redis.fetchUserAccount(userDbId);
   145	
   146	    if (!account) {
   147	      // @ts-expect-error - `{}` is a valid return for unauthenticated users. client will handle de-authenticating.
   148	      return {};
   149	    }
   150	
   151	    const { access_token, providerAccountId } = account;
   152	
   153	    const [guilds, messages] = await Promise.all([
   154	      access_token ? redis.fetchUserGuilds(providerAccountId, access_token) : ([] as UserGuild[]),
   155	      messagePromise,
   156	    ]);
   157	
   158	    if (process.env.NODE_ENV === 'development') {
   159	      guilds.push({
   160	        id: '1',
   161	        name: 'A'.repeat(32),
   162	        icon: null,
   163	        owner: true,
   164	        permissions: '0',
   165	      });
   166	    }
   167	
   168	    // `session.user` object is not the same as `params.user`
   169	    session.user.id = providerAccountId;
   170	    session.user.guilds = guilds;
   171	    session.user.preferences = user.preferences;
   172	    session.hydration = { messages };
   173	
   174	    if (preferences?.recentGuild && guilds.some(({ id }) => id === preferences.recentGuild)) {
   175	      // preload recent guild to cache in anticipation of user visiting it
   176	      redis.fetchGuild(preferences.recentGuild).catch(() => {});
   177	    }
   178	  }
   179	
   180	  return session;
   181	}

   224	        await getPostgres()
   225	          .updateTable('Accounts')

   227	            access_token: account.access_token,
   228	            refresh_token: account.refresh_token,
   229	            scope: account.scope,
   230	            expires_at: account.expires_at,
   231	          })
   232	          .where('provider', '=', 'discord')
   233	          .where('providerAccountId', '=', account.providerAccountId)
   234	          .execute();
   235	
   236	        // Update user avatar on every sign-in to keep it fresh
   237	        // profile.image is the freshly-computed avatar URL from the profile() callback
   238	        // Do NOT use user.image — it contains the stale DB value for returning users
   239	        const freshImage = (profile as { image?: string } | undefined)?.image;
   240	        if (freshImage) {
   241	          await getPostgres()
   242	            .updateTable('Users')
   243	            .set({ image: freshImage })
   244	            .where('id', '=', user.id)
   245	            .execute();
   246	        }
   247	
   248	        // Invalidate Redis cache to prevent stale tokens causing login loops
   249	        // sessionToken is not available in signIn event, so delete keys directly
   250	        await getClient().pipeline()
   251	          .del(formatKey('user-access-token', user.id))
   252	          .del(formatKey('user-guilds', account.providerAccountId))
   253	          .exec();
   254	      }
   255	    },
   256	    signOut: async (params) => {
   257	      if (params.session) {
   258	        const {
   259	          sessionToken, userId: userDbId,
   260	        } = params.session as unknown as AdapterSession;
   261	        await removeUserSession(sessionToken, userDbId);
   262	      }
   263	    },
   264	  },
