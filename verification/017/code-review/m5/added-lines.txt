+type Deferred<T> = { promise: Promise<T>, resolve: (value: T) => void, reject: (reason: unknown) => void };
+
+export function deferred<T = void>(): Deferred<T> {
+  let resolve!: (value: T) => void;
+  let reject!: (reason: unknown) => void;
+  const promise = new Promise<T>((res, rej) => { resolve = res; reject = rej; });
+  return { promise, resolve, reject };
+}

+type GeneratorAnswer = { status: number } | Error;

+  headers: Record<string, string>;
+  answer: Deferred<GeneratorAnswer>;
+  signal?: AbortSignal;
+};

+  gate: null as Deferred<void> | null,

+  queries: [] as { sql: string, parameters: readonly unknown[] }[],

+export const fakeFetch = (_url: string, init: FetchInit) => {
+  const request: PendingRequest = {
+    body: JSON.parse(init.body) as PendingRequest['body'],
+    headers: init.headers,
+    answer: deferred<GeneratorAnswer>(),
+    signal: init.signal,

+      else resolve(answer);
+    });

+type Input = Record<string, unknown>;
+type Command = { input: Input };

+function ListObjectsCommand(this: Command, input: Input) { this.input = input; }
+function DeleteObjectsCommand(this: Command, input: Input) { this.input = input; }
+function PutObjectCommand(this: Command, input: Input) { this.input = input; }

+type Sender = { send: (command: Command) => Promise<unknown> };

+function S3(this: Sender, config: Record<string, unknown>) {
+  world.s3Configs.push(config);
+  this.send = async (command: Command) => {
+    let kind = 'PutObjectCommand';
+    if (command instanceof (ListObjectsCommand as never)) kind = 'ListObjectsCommand';
+    if (command instanceof (DeleteObjectsCommand as never)) kind = 'DeleteObjectsCommand';
+    world.s3Calls.push({ kind, input: command.input });
+    if (world.gate) {
+      const { gate } = world;
+      world.gate = null;
+      await gate.promise;
+    }
+    if (kind === 'ListObjectsCommand') {
+      const prefix = command.input.Prefix as string;
+      return {
+        Contents: [...world.objects]
+          .filter(([key]) => key.startsWith(prefix))
+          .map(([Key, LastModified]) => ({ Key, LastModified })),
+      };
+    }
+    if (kind === 'DeleteObjectsCommand') {
+      (command.input.Delete as { Objects: { Key: string }[] }).Objects
+        .forEach(({ Key }) => world.objects.delete(Key));
+    }
+    return {};
+  };
