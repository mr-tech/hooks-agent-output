type Procs = { _def: { procedures: Record<string, { _def: { inputs: z.ZodTypeAny[] } }> } };
const upsertInput = (messageRouter as unknown as Procs)._def.procedures.upsert._def.inputs[0];
const MAX = 20; const STOP = Symbol('stop');
const strip = (i: z.ZodIssue) => ({ code: i.code, path: i.path, message: i.message }) as z.ZodIssue;
function capped(schema: z.ZodTypeAny) {
  return { _input: undefined, _output: undefined, async parseAsync(value: unknown) {
    const seen: z.ZodIssue[] = [];
    const errorMap: z.ZodErrorMap = (i, c) => { seen.push({ ...i, message: i.message ?? c.defaultError } as z.ZodIssue); if (seen.length > MAX) throw STOP; return { message: c.defaultError }; };
    let r;
    try { r = await schema.safeParseAsync(value, { errorMap }); } catch (e) {
      if (e !== STOP) throw e;
      throw new ZodError([...seen.slice(0, MAX).map(strip), { code: 'custom', path: [], message: `Stopped checking after ${MAX} errors` } as z.ZodIssue]);
    }
    if (!r.success) throw new ZodError(r.error.issues.map(strip));
    return r.data;
  } };
}
const router = createTRPCRouter({ save: publicProcedure.input(capped(upsertInput) as never).mutation(() => 'saved') });
const caller = router.createCaller({} as never);
const BIG = 'x'.repeat(100_000);
const rep = (n: number, f: () => unknown) => Array.from({ length: n }, f);
const btn = { type: 2, style: 5, url: 'https://a.b' };
const row = () => ({ type: 1, components: rep(5, () => ({})) });

const cont = (kids: () => unknown) => ({ type: 17, components: rep(40, kids) });

const embeds = rep(10, () => ({ fields: rep(25, () => ({})) }));
const fat = () => ({ type: 2, style: 1, emoji: { id: 1, name: 1, animated: 1 }, url: 'x', disabled: 1 });
const fatRow = () => ({ type: 1, components: rep(5, fat) });
const text = { type: 10, content: 'x' };
const inputs: [string, unknown][] = [

  ['valid', { components: [text] }],
];
describe('strip-cap probe', () => {
  it('reports', async () => {
    const out: string[] = [];
    for (const [label, data] of inputs) {
      const input = { message: { name: 'n', data } };
      const body = JSON.stringify(input).length;
      const base = upsertInput.safeParse(input);
      const t0 = performance.now();
      try { await caller.save(input as never); out.push(`${label} | body ${body} | saved`); continue; } catch (e) {
        const ms = (performance.now() - t0).toFixed(1);
        const err = e as { code?: string, message: string };
        const issues = JSON.parse(err.message) as z.ZodIssue[];
        const keys = [...new Set(issues.flatMap((i) => Object.keys(i)))].sort().join(',');
        const k = base.success ? 0 : base.error.issues.length;
        let linesEqual = 'n/a (k>20)';
        if (!base.success && k <= MAX) linesEqual = String(formatMutationError({ message: JSON.stringify(base.error.issues) }) === formatMutationError({ message: err.message }));
        out.push(`${label} | body ${body} | ${err.code} | base issues ${k} | returned ${issues.length} | message chars ${err.message.length} | carries 100k input ${err.message.includes(BIG)} | keys ${keys} | client lines equal to base ${linesEqual} | ${ms} ms`);
      }
    }
    writeFileSync(process.env.PROBE_OUT as string, `${out.join('\n')}\n`);
    expect(out.length).toBe(inputs.length);
  });
});
