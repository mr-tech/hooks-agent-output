#!/bin/sh
# usage: run-diff.sh <scratch dir>; differentials and counts at 2b82972 and 85ece66
S=$1; D=src/components/SlateEditor/utils/__tests__; OUT=$S/out; mkdir -p $OUT
for p in paste-differential keystroke-fixedpoint keystroke-settle fence-emoji-count paste-count; do for s in 2b82972 85ece66; do
  echo "$p $s load_before $(cat /proc/loadavg) $(date -u +%H:%M:%SZ)" >> $OUT/loads-diff.txt
  ( cd $S/e-$s && CRPROBE_OUT=$OUT/$p-$s.json timeout -k 10 1500 unshare -rn npx vitest run $D/crprobe.$p.test.tsx > $OUT/log-$p-$s.txt 2>&1 ); rc=$?
  echo "$p $s exit $rc load_after $(cat /proc/loadavg) $(date -u +%H:%M:%SZ)" >> $OUT/loads-diff.txt
done; done
echo done >> $OUT/loads-diff.txt
