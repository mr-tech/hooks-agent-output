
import { createEditor, Editor, Transforms } from 'slate';
import { withHistory } from 'slate-history';
import { withReact } from 'slate-react';
import { it, vi } from 'vitest';
import { writeFileSync } from 'node:fs';

import getInitialValue from '../getInitialValue';
import { withEmojis, withNormalizeNode } from '..';
import markdown from '~/utils/markdown';

import type { Descendant } from 'slate';

vi.mock('~/env.mjs', () => ({ env: {} }));
const SURFACE = markdown.interactive.parseEmoji as never;

const median = (xs: number[]) => xs.slice().sort((a, b) => a - b)[Math.floor(xs.length / 2)];
const middleOf = (n: number) => Math.floor(n / 2);

const newEditor = (text: string) => {
  const editor = withNormalizeNode(withEmojis(withHistory(withReact(createEditor()))), false, SURFACE);
  editor.children = getInitialValue(text, SURFACE, false) as Descendant[];
  return editor;
};

const backspace = (e: Editor) => { e.deleteBackward('character'); };
const typeTick = (e: Editor) => { e.insertText('`'); };

const FENCE: Array<[string, (n: number) => Record<number, string>, (e: Editor, n: number) => void]> = [
  ['third backtick typed at the end of line 2', () => ({ 2: '``' }), (e) => { Transforms.select(e, Editor.end(e, [2])); typeTick(e); }],
  ['Backspace at the end of the ``` on line 2', () => ({ 2: '```' }), (e) => { Transforms.select(e, Editor.end(e, [2])); backspace(e); }],
  ['third backtick at the end of the closing delimiter in the middle', (n) => ({ 2: '```', [middleOf(n)]: '``' }), (e, n) => { Transforms.select(e, Editor.end(e, [middleOf(n)])); typeTick(e); }],
  ['Backspace at the end of the closing ``` in the middle', (n) => ({ 2: '```', [middleOf(n)]: '```' }), (e, n) => { Transforms.select(e, Editor.end(e, [middleOf(n)])); backspace(e); }],
];

it('timed readings', { timeout: 3000000 }, () => {
  const rows: object[] = [];
  const fills = ['a', 'a🙂'];
  for (const fill of fills) {
    for (const n of [200, 1000]) {
      const reps = n === 200 ? 5 : 3;
      for (const [name, own, act] of FENCE) {
        if (fill === 'a' && n === 1000 && false) continue;
        const text = Array.from({ length: n }, (_, i) => (own(n)[i] ?? fill)).join('\n');
        const times: number[] = []; let error = '';
        for (let k = 0; k < reps; k += 1) {
          const editor = newEditor(text);
          const t0 = performance.now();
          try { act(editor, n); } catch (e) { error = String(e).slice(0, 70); }
          times.push(performance.now() - t0);
        }
        rows.push({ member: `fence: ${name}`, fill, n, reps, medianMs: +median(times).toFixed(1), minMs: +Math.min(...times).toFixed(1), maxMs: +Math.max(...times).toFixed(1), error, load: require('node:fs').readFileSync('/proc/loadavg', 'utf8').split(' ').slice(0, 3).join(' ') });
      }
      const times: number[] = []; let error = '';
      for (let k = 0; k < reps; k += 1) {
        const editor = newEditor('');
        editor.selection = { anchor: Editor.start(editor, []), focus: Editor.start(editor, []) };
        const text = Array.from({ length: n }, () => fill).join('\n');
        const data = { getData: (f: string) => (f === 'text/plain' ? text : ''), types: ['text/plain'], files: [], items: [] } as unknown as DataTransfer;
        const t0 = performance.now();
        try { (editor as unknown as { insertData: (d: DataTransfer) => void }).insertData(data); } catch (e) { error = String(e).slice(0, 70); }
        times.push(performance.now() - t0);
      }
      rows.push({ member: 'paste', fill, n, reps, medianMs: +median(times).toFixed(1), minMs: +Math.min(...times).toFixed(1), maxMs: +Math.max(...times).toFixed(1), error, load: require('node:fs').readFileSync('/proc/loadavg', 'utf8').split(' ').slice(0, 3).join(' ') });
    }
  }
  writeFileSync(process.env.PROBE_OUT as string, rows.map((r) => JSON.stringify(r)).join('\n'));
});
