import { beforeAll, it, vi } from 'vitest';
import { appendFileSync } from 'node:fs';
import { Editor, Transforms } from 'slate';
import { editorHolding } from '../../__tests__/helpers/keystrokes';
import { FENCE_KEYSTROKES } from '../../__tests__/helpers/fenceAndQuoteKeystrokes';
import { startEngine } from '~/utils/markdown/engine';
vi.mock('~/env.mjs', () => ({ env: {} }));
beforeAll(async () => { await startEngine(); });
const median = (xs: number[]) => xs.slice().sort((a, b) => a - b)[Math.floor(xs.length / 2)];
const row = (o: object) => appendFileSync(process.env.PROBE_OUT as string, `${JSON.stringify(o)}\n`);
it('class cost', { timeout: 900000 }, () => {
  // typographic quote lines, close a fence on line 2
  for (const fill of ['a’', 'a’🙂']) for (const n of [200, 1000]) {
    const k = FENCE_KEYSTROKES[1];
    const ts: number[] = [];
    for (let r = 0; r < 3; r += 1) { const e = editorHolding(k.text(n, fill), 'embedDescription'); const t0 = performance.now(); k.press(e, n); ts.push(performance.now() - t0); }
    row({ probe: 'typographic quotes, Backspace on ``` of line 2', fill, n, medianMs: +median(ts).toFixed(1) });
  }
  // a lone backtick typed at the end of line 0 with a closing backtick on the last line: emoji lines between turn to text
  for (const fill of ['a🙂']) for (const n of [200, 1000]) {
    const ts: number[] = []; let ops = 0;
    for (let r = 0; r < 3; r += 1) {
      const text = Array.from({ length: n }, (_, i) => (i === n - 1 ? `${fill}\`` : fill)).join('\n');
      const e = editorHolding(text, 'embedDescription');
      Transforms.select(e, Editor.end(e, [0]));
      const o0 = e.operations.length; const t0 = performance.now(); let err = ''; try { e.insertText('`'); } catch (x) { err = String(x).slice(0, 60); } ts.push(performance.now() - t0); ops = e.operations.length - o0; if (err) row({ error: err, n });
    }
    row({ probe: 'backtick at end of line 0, closing backtick on the last line', fill, n, medianMs: +median(ts).toFixed(1), ops });
  }
});
