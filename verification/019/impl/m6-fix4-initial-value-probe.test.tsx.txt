import { beforeAll, it, vi } from 'vitest';
import { writeFileSync, readFileSync } from 'node:fs';
import getInitialValue from '../getInitialValue';
import { startEngine } from '~/utils/markdown/engine';

vi.mock('~/env.mjs', () => ({ env: {} }));
beforeAll(async () => { await startEngine(); });
const median = (xs: number[]) => xs.slice().sort((a, b) => a - b)[Math.floor(xs.length / 2)];

it('getInitialValue timed', { timeout: 600000 }, () => {
  const rows: object[] = [];
  for (const surface of ['embedDescription', 'content'] as const) {
    for (const fill of ['a', 'a🙂']) {
      for (const n of [200, 1000]) {
        const text = Array.from({ length: n }, () => fill).join('\n');
        const v = getInitialValue(text, surface, false) as any[];
        const emoji = v.reduce((a: number, l: any) => a + l.children.filter((c: any) => c.type === 'emoji').length, 0);
        const ts: number[] = [];
        for (let k = 0; k < 7; k += 1) {
          const t0 = performance.now();
          getInitialValue(text, surface, false);
          ts.push(performance.now() - t0);
        }
        rows.push({ emoji, lines: v.length, surface, fill, n, medianMs: +median(ts).toFixed(1), minMs: +Math.min(...ts).toFixed(1), maxMs: +Math.max(...ts).toFixed(1), firstMs: +ts[0].toFixed(1), load: readFileSync('/proc/loadavg', 'utf8').split(' ').slice(0, 3).join(' ') });
      }
    }
  }
  writeFileSync(process.env.PROBE_OUT as string, rows.map((r) => JSON.stringify(r)).join('\n') + '\n');
});
