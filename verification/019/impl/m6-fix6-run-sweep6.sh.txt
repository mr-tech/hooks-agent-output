#!/bin/sh
# usage: run-sweep6.sh <mutants> <work> <out> <sha>; the self-check sweep at <sha> in its own 5G no-swap scope, one worker, no network,
# the SlateEditor suite per mutant, 250 s per-mutant timeout
mkdir -p $2 $3
cd /usr/local/bin/hooks/website && systemd-run --user --scope -q -p MemoryMax=5G -p MemorySwapMax=0 unshare -rn python3 ~/.claude/tools/mutation-run.py --repo /usr/local/bin/hooks/website --sha $4 \
  --mutants $1 --work $2 --out $3 --timeout 250 --dep node_modules -- npx vitest run --project node --project dom --maxWorkers=1 src/components/SlateEditor > $3.log.txt 2>&1
echo "exit $?" >> $3.log.txt
